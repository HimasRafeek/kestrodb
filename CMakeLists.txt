cmake_minimum_required(VERSION 3.20)
project(KestroDB VERSION 0.9.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
# The static BLAKE3 library is also linked into the Node addon (a shared module).
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "KestroDB supports 64-bit targets only")
endif()
if(NOT WIN32 AND NOT APPLE AND NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "Supported platform backends: Windows, Linux, macOS")
endif()
option(KESTRO_BUILD_TESTS "Build native tests and fault-injection worker" ON)
option(KESTRO_PORTABLE_HASH "Disable architecture-specific BLAKE3 acceleration" OFF)
option(KESTRO_STATIC_MINGW "Link MinGW runtime libraries statically" ON)
option(KESTRO_NODE_ADDON "Build the in-process Node-API addon (kestro.node)" ON)
# Linux release builds (scripts/build-linux.sh): binaries that run on other distributions.
# KESTRO_STATIC_RUNTIME links libstdc++ statically, its symbols hidden (so the addon never mixes
# its C++ runtime with the one node or python loaded); libgcc_s stays shared: it is on every glibc
# system, and GCC 12's static unwinder would need glibc 2.35 (_dl_find_object). KESTRO_STATIC_CRYPTO
# links libcrypto (only SHA-256 is used) statically and hidden into kestro.node, the C library
# and the worker; kestro-server keeps the system OpenSSL 3 (TLS, security updates by the OS).
option(KESTRO_STATIC_RUNTIME "Linux: link libstdc++ statically (hidden)" OFF)
option(KESTRO_STATIC_CRYPTO "Linux: link libcrypto statically (hidden) into everything but kestro-server" OFF)
# Static MSVC runtime: the worker and addon run without the VC++ redistributable.
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(KESTRO_SANITIZER "" CACHE STRING "Optional address or thread sanitizer")
set(KESTRO_OUTPUT_DIR "${CMAKE_BINARY_DIR}/bin" CACHE PATH "Executable directory")
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${KESTRO_OUTPUT_DIR}")
foreach(config DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
  set(CMAKE_RUNTIME_OUTPUT_DIRECTORY_${config} "${KESTRO_OUTPUT_DIR}")
endforeach()
if(KESTRO_SANITIZER)
  if(MSVC OR WIN32)
    message(FATAL_ERROR "Sanitizer option is provided for Unix GCC/Clang builds")
  endif()
  if(KESTRO_SANITIZER STREQUAL "address")
    set(sanitizers address,undefined)
  elseif(KESTRO_SANITIZER STREQUAL "thread")
    set(sanitizers thread)
  else()
    message(FATAL_ERROR "Unknown sanitizer")
  endif()
  add_compile_options(-fsanitize=${sanitizers} -fno-omit-frame-pointer -g)
  add_link_options(-fsanitize=${sanitizers})
endif()
set(b "${CMAKE_CURRENT_SOURCE_DIR}/vendor/blake3")
add_library(blake3 STATIC "${b}/blake3.c" "${b}/blake3_dispatch.c" "${b}/blake3_portable.c")
target_include_directories(blake3 PUBLIC "${b}")
string(TOLOWER "${CMAKE_SYSTEM_PROCESSOR}" cpu)
if(MSVC AND CMAKE_VS_PLATFORM_NAME)
  string(TOLOWER "${CMAKE_VS_PLATFORM_NAME}" cpu)
elseif(APPLE AND CMAKE_OSX_ARCHITECTURES)
  list(LENGTH CMAKE_OSX_ARCHITECTURES arch_count)
  if(arch_count GREATER 1 AND NOT KESTRO_PORTABLE_HASH)
    message(FATAL_ERROR "Universal macOS builds require KESTRO_PORTABLE_HASH=ON, or build each architecture separately")
  elseif(arch_count EQUAL 1)
    string(TOLOWER "${CMAKE_OSX_ARCHITECTURES}" cpu)
  endif()
endif()
if(KESTRO_PORTABLE_HASH)
  target_compile_definitions(blake3 PRIVATE BLAKE3_NO_SSE2 BLAKE3_NO_SSE41 BLAKE3_NO_AVX2 BLAKE3_NO_AVX512 BLAKE3_USE_NEON=0)
elseif(cpu MATCHES "^(x86_64|amd64|x64)$")
  if(MSVC)
    enable_language(ASM_MASM)
    set(asm_suffix windows_msvc.asm)
  else()
    enable_language(ASM)
    if(WIN32)
      set(asm_suffix windows_gnu.S)
    else()
      set(asm_suffix unix.S)
    endif()
  endif()
  foreach(simd sse2 sse41 avx2 avx512)
    target_sources(blake3 PRIVATE "${b}/blake3_${simd}_x86-64_${asm_suffix}")
  endforeach()
elseif(cpu MATCHES "^(aarch64|arm64)$")
  if(MSVC)
    # A portable fallback also supports MSVC ARM64 without GNU NEON headers.
    target_compile_definitions(blake3 PRIVATE BLAKE3_USE_NEON=0)
  else()
    target_sources(blake3 PRIVATE "${b}/blake3_neon.c")
    target_compile_definitions(blake3 PRIVATE BLAKE3_USE_NEON=1)
  endif()
else()
  message(FATAL_ERROR "Unknown CPU; set KESTRO_PORTABLE_HASH=ON for a portable build")
endif()
add_library(kestro INTERFACE)
target_include_directories(kestro INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/src")
target_link_libraries(kestro INTERFACE blake3)
if(WIN32)
  configure_file(cmake/windows.rc.in "${CMAKE_BINARY_DIR}/windows.rc" @ONLY)
  target_compile_definitions(kestro INTERFACE _WIN32_WINNT=0x0A00 UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_link_libraries(kestro INTERFACE bcrypt)
  if(MINGW AND KESTRO_STATIC_MINGW)
    target_link_options(kestro INTERFACE -static -static-libgcc -static-libstdc++)
  endif()
else()
  find_package(Threads REQUIRED)
  target_link_libraries(kestro INTERFACE Threads::Threads)
  target_compile_definitions(kestro INTERFACE _FILE_OFFSET_BITS=64)
  if(NOT APPLE)
    find_package(OpenSSL 3 REQUIRED COMPONENTS Crypto)
    if(KESTRO_STATIC_CRYPTO)
      get_filename_component(crypto_dir "${OPENSSL_CRYPTO_LIBRARY}" DIRECTORY)
      find_library(KESTRO_CRYPTO_ARCHIVE NAMES libcrypto.a HINTS "${crypto_dir}" REQUIRED)
      add_library(kestro_static_crypto INTERFACE)
      target_include_directories(kestro_static_crypto INTERFACE "${OPENSSL_INCLUDE_DIR}")
      target_link_libraries(kestro_static_crypto INTERFACE "${KESTRO_CRYPTO_ARCHIVE}" ${CMAKE_DL_LIBS})
      target_link_options(kestro_static_crypto INTERFACE "LINKER:--exclude-libs,libcrypto.a")
      set(kestro_crypto kestro_static_crypto)
    else()
      target_link_libraries(kestro INTERFACE OpenSSL::Crypto)
    endif()
  endif()
  if(KESTRO_STATIC_RUNTIME AND NOT APPLE)
    target_link_options(kestro INTERFACE -static-libstdc++ "LINKER:--exclude-libs,ALL")
  endif()
endif()
if(MSVC)
  target_compile_options(kestro INTERFACE /utf-8 /EHsc /W4)
  target_compile_options(blake3 PRIVATE "$<$<COMPILE_LANGUAGE:C>:/utf-8>")
endif()
function(worker name)
  add_executable(${name} src/worker.cpp)
  if(MSVC)
    target_link_options(${name} PRIVATE "/MANIFEST:EMBED" "/MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/cmake/windows.manifest")
  elseif(WIN32)
    target_sources(${name} PRIVATE "${CMAKE_BINARY_DIR}/windows.rc")
  endif()
  target_link_libraries(${name} PRIVATE kestro ${kestro_crypto})
  if(MINGW)
    target_link_options(${name} PRIVATE -municode)
  endif()
endfunction()
worker(kestro-worker)
# C API shared library (src/kestrodb.h) for every other language: PHP FFI, Python ctypes,
# Go cgo, Java, C#, Rust... Reads in-process alongside kestro-server, or embedded owner mode.
add_library(kestrodb SHARED src/capi.cpp)
target_link_libraries(kestrodb PRIVATE kestro ${kestro_crypto} ${CMAKE_DL_LIBS})
set_target_properties(kestrodb PROPERTIES CXX_VISIBILITY_PRESET hidden
  RUNTIME_OUTPUT_DIRECTORY "${KESTRO_OUTPUT_DIR}" LIBRARY_OUTPUT_DIRECTORY "${KESTRO_OUTPUT_DIR}")
foreach(config DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
  set_target_properties(kestrodb PROPERTIES RUNTIME_OUTPUT_DIRECTORY_${config} "${KESTRO_OUTPUT_DIR}" LIBRARY_OUTPUT_DIRECTORY_${config} "${KESTRO_OUTPUT_DIR}")
endforeach()
if(MINGW)
  set_target_properties(kestrodb PROPERTIES PREFIX "")
endif()
# Network server: one database shared by many clients (PHP, Python, Node, ...).
add_executable(kestro-server src/server.cpp)
target_link_libraries(kestro-server PRIVATE kestro)
target_compile_definitions(kestro-server PRIVATE KESTRO_VERSION="${PROJECT_VERSION}")
if(KESTRO_STATIC_CRYPTO)
  target_link_libraries(kestro-server PRIVATE OpenSSL::Crypto)
endif()
# TLS (--tls-cert/--tls-key) where OpenSSL's TLS library is available (Linux, macOS with
# OpenSSL); on Windows through Schannel (src/tls_windows.hpp), part of the system.
if(NOT WIN32)
  find_package(OpenSSL 3 COMPONENTS SSL)
  if(TARGET OpenSSL::SSL)
    target_link_libraries(kestro-server PRIVATE OpenSSL::SSL)
    target_compile_definitions(kestro-server PRIVATE KESTRO_TLS=1)
  endif()
endif()
if(WIN32)
  target_link_libraries(kestro-server PRIVATE ws2_32 secur32 crypt32 ncrypt)
  target_compile_definitions(kestro-server PRIVATE KESTRO_WINTLS=1)
  if(MSVC)
    target_link_options(kestro-server PRIVATE "/MANIFEST:EMBED" "/MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/cmake/windows.manifest")
  else()
    target_sources(kestro-server PRIVATE "${CMAKE_BINARY_DIR}/windows.rc")
    target_link_options(kestro-server PRIVATE -municode)
  endif()
endif()
if(KESTRO_NODE_ADDON)
  add_library(kestro-node MODULE src/addon.cpp)
  set_target_properties(kestro-node PROPERTIES PREFIX "" OUTPUT_NAME kestro SUFFIX ".node"
    LIBRARY_OUTPUT_DIRECTORY "${KESTRO_OUTPUT_DIR}" CXX_VISIBILITY_PRESET hidden)
  foreach(config DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
    set_target_properties(kestro-node PROPERTIES LIBRARY_OUTPUT_DIRECTORY_${config} "${KESTRO_OUTPUT_DIR}")
  endforeach()
  target_include_directories(kestro-node PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/vendor/node-api/include")
  target_link_libraries(kestro-node PRIVATE kestro ${kestro_crypto})
  if(WIN32)
    # Import library for the Node-API symbols exported by node.exe, built from a .def list.
    set(node_def "${CMAKE_CURRENT_SOURCE_DIR}/cmake/node_api.def")
    if(MSVC)
      if(cpu MATCHES "^(aarch64|arm64)$")
        set(node_machine ARM64)
      else()
        set(node_machine X64)
      endif()
      set(node_lib "${CMAKE_BINARY_DIR}/node_api.lib")
      add_custom_command(OUTPUT "${node_lib}" COMMAND "${CMAKE_AR}" /nologo "/def:${node_def}" "/out:${node_lib}" "/machine:${node_machine}" DEPENDS "${node_def}" VERBATIM)
    else()
      set(node_lib "${CMAKE_BINARY_DIR}/libnode_api.a")
      find_program(KESTRO_DLLTOOL NAMES ${CMAKE_C_COMPILER_TARGET}-dlltool x86_64-w64-mingw32-dlltool dlltool REQUIRED)
      add_custom_command(OUTPUT "${node_lib}" COMMAND "${KESTRO_DLLTOOL}" -d "${node_def}" -l "${node_lib}" -D node.exe DEPENDS "${node_def}" VERBATIM)
    endif()
    add_custom_target(kestro-node-imports DEPENDS "${node_lib}")
    add_dependencies(kestro-node kestro-node-imports)
    target_link_libraries(kestro-node PRIVATE "${node_lib}")
  elseif(APPLE)
    target_link_options(kestro-node PRIVATE -undefined dynamic_lookup)
  endif()
endif()

  worker(kestro-worker-faults)
  target_compile_definitions(kestro-worker-faults PRIVATE KESTRO_TEST_FAULTS)
  add_executable(kestro-portable-tests src/tests_portable.cpp)

  add_test(NAME portable COMMAND kestro-portable-tests)
  set_tests_properties(portable PROPERTIES TIMEOUT 900)  # the concurrency tests run 10x slower under the thread sanitizer
  # Simulated power loss: every file operation is recorded and disk states a power cut could

  add_executable(kestro-powerloss-tests src/tests_powerloss.cpp)

    target_link_options(kestro-powerloss-tests PRIVATE "/MANIFEST:EMBED" "/MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/cmake/windows.manifest")

  target_link_libraries(kestro-powerloss-tests PRIVATE kestro ${kestro_crypto})
  add_test(NAME powerloss COMMAND kestro-powerloss-tests)
  set_tests_properties(powerloss PROPERTIES TIMEOUT 600)
  if(NOT WIN32)
    add_executable(kestro-posix-tests src/tests.cpp)
    target_compile_definitions(kestro-posix-tests PRIVATE KESTRO_TEST_FAULTS)
    target_link_libraries(kestro-posix-tests PRIVATE kestro ${kestro_crypto})
    add_test(NAME posix COMMAND kestro-posix-tests)
    set_tests_properties(posix PROPERTIES TIMEOUT 180)
  endif()
endif()
